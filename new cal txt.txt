  <script>
    const originalDate = document.getElementById('originalDate');
    const originalTime = document.getElementById('originalTime');
    const newDate = document.getElementById('newDate');
    const newTime = document.getElementById('newTime');
    const result = document.getElementById('result');
    const difference = document.getElementById('difference');
    const description = document.getElementById('description');
    const copyButton = document.getElementById('copyButton');

    let copyText = '';

    function today() {
      const date = new Date();
      const year = date.getFullYear();
      const month = String(date.getMonth() + 1).padStart(2, '0');
      const day = String(date.getDate()).padStart(2, '0');
      return year + '-' + month + '-' + day;
    }

    function normalizeTime(value) {
      const clean = value.trim().replace(/\s/g, '');

      if (/^\d{1,2}:\d{2}$/.test(clean)) {
        const parts = clean.split(':');
        return parts[0].padStart(2, '0') + ':' + parts[1];
      }

      if (/^\d{3,4}$/.test(clean)) {
        const padded = clean.padStart(4, '0');
        return padded.slice(0, 2) + ':' + padded.slice(2);
      }

      return clean;
    }

    function parseTime(value) {
      const normalized = normalizeTime(value);
      const match = normalized.match(/^([01]\d|2[0-3]):([0-5]\d)$/);
      if (!match) return null;

      return {
        normalized: normalized,
        hour: Number(match[1]),
        minute: Number(match[2])
      };
    }

    function parseSchedule(dateValue, timeValue) {
      const time = parseTime(timeValue);
      if (!dateValue || !time) return null;

      const dateParts = dateValue.split('-').map(Number);
      return {
        date: dateValue,
        time: time.normalized,
        value: Date.UTC(dateParts[0], dateParts[1] - 1, dateParts[2], time.hour, time.minute)
      };
    }

    function formatDate(value) {
      const parts = value.split('-').map(Number);
      const date = new Date(Date.UTC(parts[0], parts[1] - 1, parts[2]));
      return new Intl.DateTimeFormat('en-US', {
        month: 'short',
        day: 'numeric',
        year: 'numeric',
        timeZone: 'UTC'
      }).format(date);
    }

    function durationWords(totalMinutes) {
      const days = Math.floor(totalMinutes / 1440);
      const remainder = totalMinutes % 1440;
      const hours = Math.floor(remainder / 60);
      const minutes = remainder % 60;
      const parts = [];

      if (days) parts.push(days + ' day' + (days === 1 ? '' : 's'));
      if (hours) parts.push(hours + ' hour' + (hours === 1 ? '' : 's'));
      if (minutes) parts.push(minutes + ' minute' + (minutes === 1 ? '' : 's'));

      if (!parts.length) return '0 minutes';
      if (parts.length === 1) return parts[0];
      return parts.slice(0, -1).join(', ') + ' and ' + parts[parts.length - 1];
    }

    function shortDuration(totalMinutes) {
      const days = Math.floor(totalMinutes / 1440);
      const remainder = totalMinutes % 1440;
      const hours = Math.floor(remainder / 60);
      const minutes = remainder % 60;

      if (days) return days + 'd ' + hours + 'h ' + minutes + 'm';
      return hours + ':' + String(minutes).padStart(2, '0');
    }

    function calculate() {
      const oldSchedule = parseSchedule(originalDate.value, originalTime.value);
      const updatedSchedule = parseSchedule(newDate.value, newTime.value);

      originalDate.classList.toggle('invalid', !originalDate.value);
      newDate.classList.toggle('invalid', !newDate.value);
      originalTime.classList.toggle('invalid', !parseTime(originalTime.value));
      newTime.classList.toggle('invalid', !parseTime(newTime.value));

      if (!oldSchedule || !updatedSchedule) {
        result.className = 'result invalid';
        difference.textContent = 'Incomplete';
        description.textContent = 'Enter valid dates and times.';
        copyButton.disabled = true;
        copyText = '';
        return;
      }

      const minutes = Math.round((updatedSchedule.value - oldSchedule.value) / 60000);
      const absolute = Math.abs(minutes);
      const direction = minutes > 0 ? 'later' : minutes < 0 ? 'earlier' : 'unchanged';

      if (direction === 'unchanged') {
        result.className = 'result unchanged';
        difference.textContent = '0:00';
        description.textContent = 'No schedule change.';
      } else {
        result.className = 'result ' + direction;
        difference.textContent = (minutes > 0 ? '+' : '-') + shortDuration(absolute);
        description.textContent = durationWords(absolute) + ' ' + direction + '.';
      }

      copyText = [
        'Original: ' + formatDate(oldSchedule.date) + ' at ' + oldSchedule.time,
        'New: ' + formatDate(updatedSchedule.date) + ' at ' + updatedSchedule.time,
        'Change: ' + description.textContent
      ].join('\n');

      copyButton.disabled = false;
    }

    function formatTime(input) {
      const parsed = parseTime(input.value);
      if (parsed) input.value = parsed.normalized;
      calculate();
    }

    [originalDate, newDate].forEach(input => {
      input.addEventListener('change', calculate);
    });

    [originalTime, newTime].forEach(input => {
      input.addEventListener('input', calculate);
      input.addEventListener('blur', () => formatTime(input));
      input.addEventListener('keydown', event => {
        if (event.key === 'Enter') formatTime(input);
      });
    });

    document.getElementById('swapButton').addEventListener('click', () => {
      const savedDate = originalDate.value;
      const savedTime = originalTime.value;

      originalDate.value = newDate.value;
      originalTime.value = newTime.value;
      newDate.value = savedDate;
      newTime.value = savedTime;
      calculate();
    });

    document.getElementById('clearButton').addEventListener('click', () => {
      const currentDate = today();
      originalDate.value = currentDate;
      newDate.value = currentDate;
      originalTime.value = '00:00';
      newTime.value = '00:00';
      calculate();
    });

    copyButton.addEventListener('click', async () => {
      if (!copyText) return;

      try {
        await navigator.clipboard.writeText(copyText);
      } catch (error) {
        const area = document.createElement('textarea');
        area.value = copyText;
        document.body.appendChild(area);
        area.select();
        document.execCommand('copy');
        area.remove();
      }

      copyButton.textContent = 'Copied';
      setTimeout(() => copyButton.textContent = 'Copy result', 1500);
    });

    originalDate.value = today();
    newDate.value = today();
    calculate();
  </script>
